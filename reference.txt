from langchain_google_genai import ChatGoogleGenerativeAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

llm = ChatGoogleGenerativeAI(model="gemini-3.5-flash")
out = StrOutputParser()

def transform_case(result :str):
    return result.upper()

prompts = ChatPromptTemplate.from_messages([
    {"role": "system", "content": "you are a translator and translate input in {language}"},
    {"role": "user", "content": "{query}"}
])

#final_prompt = prompts.format_prompt(language="Python", shape="pyramid")

#res = llm.invoke(final_prompt)

chain = prompts | llm | out | transform_case

res = chain.invoke({"language": "Hinglish", "query": "Hello, how are you?"})

"Email is vinodrathod@gmail.com"

           Email: vinodrathod@gmail.com

from pydantic import BaseModel, Field
class ResponseStrecture(BaseModel):
    name:str = Field(description="Complete name")
    email:str = Field(description="Email address")
    age:int = Field(description="Age of the person")

strectured_llm = llm.with_structured_output(ResponseStrecture)

res = strectured_llm.invoke(f"Please give me only name , email and age from the text {text}")

res.model_dump()

# output : {'name': 'Vinod Rathod', 'email': 'vinodrathod@gmail.com', 'age': 0}

class Movies(BaseModel):
    title:str = Field(description="Movie title")
    year:int = Field(description="Release year")

class AllMovies(BaseModel):
    movies:List[Movies] 

movies_llm = llm.with_structured_output(AllMovies)

res = movies_llm.invoke("Give me three movies")

#  output : {'movies': [
        {'title': 'Inception', 'year': 2010}, 
        {'title': 'The Shawshank Redemption', 'year': 1994}, 
        {'title': 'The Godfather', 'year': 1972}
#  ]}
